# --- Core Web Server ---
fastapi               # Framework web (include starlette e pydantic)
uvicorn               # Server ASGI (avvio locale e Docker)
jinja2                # Template HTML (Jinja2Templates)
python-multipart      # Necessario per gestire i form di login e invio punti
pydantic-settings     # Configurazione da .env

# --- Database ---
sqlmodel              # L'unione tra SQLAlchemy e Pydantic
psycopg2-binary       # Driver PostgreSQL (Neon/Supabase)

# --- Security & Auth ---
python-jose           # Per la creazione e validazione dei JWT (HS256, backend Python puro)

# --- Utility ---
requests              # Per lo script di smoke test in produzione
httpx                 # Client async per l'API Ecologi + TestClient dei test

# --- Test ---
pytest                # Test suite (gira su SQLite in-memory, non tocca il DB reale)

# --- Production ---
gunicorn              # Process manager per far girare FastAPI in produzione
